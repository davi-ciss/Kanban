 Converte um range (1 célula ou bloco colado) já restrito à coluna JIRA em hiperlinks
 (mostra só o código, cor azul padrão). Chamável também pela integração (ver nota no fim).
function processarLinkJira_(range){
  var vals = range.getValues();           valores atuais do range (matriz linhas x colunas)
  var sheet = range.getSheet();
  var row0 = range.getRow(), col0 = range.getColumn();  canto superior-esquerdo (1-based)

  for(var i=0;ivals.length;i++){          percorre cada linha do range
    for(var j=0;jvals[i].length;j++){     percorre cada coluna do range (normalmente só 1)
      var v = vals[i][j];
      if(typeof v !== 'string') continue;  ignora númerodatavazio
      v = v.trim();
      if(!v) continue;

      var ticket=null, url=null;
      var m = v.match(FE_JIRA_URL_RE);     1ª tentativa célula tem um link colado
      if(m){
        ticket = m[2].toUpperCase();                código extraído do link
        url = 'https' + m[1] + 'browse' + ticket;  usa o host do próprio link
      } else if(FE_JIRA_CODE_RE.test(v)){  2ª tentativa só o código, sem link
        ticket = v.toUpperCase();
        url = 'https' + FE_JIRA_HOST_PADRAO + 'browse' + ticket;  usa host padrão
      } else {
        continue;  não é link nem código reconhecível — não mexe na célula
      }

       Sem setTextStyle - mantém a corsublinhado azul padrão de hiperlink do Sheets
      sheet.getRange(row0+i, col0+j).setRichTextValue(
        SpreadsheetApp.newRichTextValue().setText(ticket).setLinkUrl(url).build()
      );
    }
  }
}

  var s = e.range.getSheet();
  if(e.range.getRow()=1) return;  edição no cabeçalho ignora

   (1) Coluna JIRA de QUALQUER aba - hiperlink ABC-123

  if(jiraCol=0){
    var cj = jiraCol+1;                                         1-based
    var ini = e.range.getColumn(), fim = ini + e.range.getNumColumns() - 1;  faixa de colunas editada
    if(cj=ini && cj=fim){                                     a edição tocou a coluna Jira
      processarLinkJira_(s.getRange(e.range.getRow(), cj, e.range.getNumRows(), 1));  restringe à coluna Jira
    }

   (2) Coluna TICKET (Zendesk) só na aba Apoios - hiperlink com o número
  if(s.getName()===Apoios){

      if(ct=ini2 && ct=fim2){
        var rngT = s.getRange(e.range.getRow(), ct, e.range.getNumRows(), 1);
        var valsT = rngT.getValues();

            SpreadsheetApp.newRichTextValue().setText(mz[1]).setLinkUrl(vt.trim()).build());
        }
      }
    }
  }
}
